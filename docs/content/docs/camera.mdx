[`useCameraFrame`](/api/functions/useCameraFrame) turns a VisionCamera frame into
a [`TgfxImage`](/api/interfaces/TgfxImage) and a
[`CameraFramePosture`](/api/interfaces/CameraFramePosture), together a
[`CameraFrameSource`](/api/interfaces/CameraFrameSource).

<Tab value="Declarative">

```tsx title="Preview.tsx"
import { useCamera, useFrameOutput } from 'react-native-vision-camera'
import { Canvas, CameraFrame, useCameraFrame } from 'react-native-tgfx'

export function Preview() {
  const { image, posture, onFrame } = useCameraFrame()

  const frameOutput = useFrameOutput({
    pixelFormat: 'yuv',
    dropFramesWhileBusy: true,
    enablePhysicalBufferRotation: true,
    // Headless: this output is the only one, so deferring it behind a preview starts it late.
    allowDeferredStart: false,
    onFrame,
  })

  useCamera({
    isActive: true,
    device: 'back',
    outputs: [frameOutput],
  })

  return (
    <Canvas style={StyleSheet.absoluteFill} opaque={true}>
      <CameraFrame image={image} {...posture} />
    </Canvas>
  )
}
```

```ts
import { Scene, cameraFrame, createCameraFrameSink, createHandle } from 'react-native-tgfx'

const scene = Scene.create()
const node = createHandle()

const sink = createCameraFrameSink({
  // sink.image is null until the first frame mints the handle - build the node here.
  onImage: (image) => scene.add(cameraFrame({ ref: node, image, ...sink.posture })),
  onPosture: (posture) => node.set(posture),
})

// VisionCamera side, unchanged apart from the pump: onFrame: sink.push
const frameOutput = useFrameOutput({
  pixelFormat: 'yuv',
  dropFramesWhileBusy: true,
  enablePhysicalBufferRotation: true,
  allowDeferredStart: false,
  onFrame: sink.push,
})

scene.attach(viewRef)
// sink.dispose() when the stream ends - nothing else frees the minted handle.
```

Get camera permission first. No camera view is mounted. Keep
`allowDeferredStart: false`, or frames start late.

Keep `useFrameOutput` options stable (a module constant or memoised).

## Zero copy

[`imageFromNativeBuffer`](/api/functions/imageFromNativeBuffer) imports the
frame's buffer ([`NativeBufferLike`](/api/interfaces/NativeBufferLike)) as a
texture with no copy. Later frames reuse the handle:

```ts
handle.value == null
  ? imageFromNativeBuffer(buffer.pointer, format)   // first frame
  : handle.value.setNativeBuffer(buffer.pointer, format)  // every frame after
```

[`setNativeBuffer`](/api/interfaces/TgfxImage) releases the previous buffer
before returning.

<Callout type="warn">
Don't write into a buffer a handle points at.
</Callout>

`useCameraFrame` manages this for you.

<Callout type="warn">
The frame in your callback dies when it returns. Don't retain it, call
`getNativeBuffer()` again, or store its pointer.
</Callout>

## No JS per frame

```bash
npm install react-native-tgfx-vision-camera
```

`useTgfxCameraSink` does the import natively, so frames skip JS. Attach it once
the handle exists:

```tsx
const sink = useTgfxCameraSink()

useEffect(() => {
  if (image != null) sink.attach(frameOutput, image.handle)
}, [sink, frameOutput, image])
```

After `attach`, orientation stops reaching JS, so keep
`enablePhysicalBufferRotation: true`. Not for frame processors that run JS,
like a detector.

## Zooming

The [`viewport`](/docs/animation#zoom-and-pan) doesn't move a `<CameraFrame>`.
Wrap it in `<Group transform>` to zoom it.

## Face blur

Allocate fixed slots. Zero-size slots draw nothing.

<Tabs items={['Declarative', 'Imperative']}>
<Tab value="Declarative">

```tsx
const SLOTS = [0, 1, 2, 3]
const [ovals] = useState(() => SLOTS.map(() => createHandle()))

const detect = useCallback((frame: Frame) => {
  'worklet'
  const faces = detector.detectFaces(frame)
  for (const i of SLOTS) {
    const b = faces[i]?.bounds
    ovals[i]!.set({
      x: b?.x ?? 0,
      y: b?.y ?? 0,
      width: b?.width ?? 0,
      height: b?.height ?? 0,
    })
  }
}, [detector, ovals])

return { ...useCameraFrame(detect), ovals }
```

```tsx
<CameraFrame image={image} {...posture}>
  {SLOTS.map((i) => (
    <Oval key={i} ref={ovals[i]} x={0} y={0} width={0} height={0}>
      <Paint>
        <ImageShader image={image} />
        <Blur blur={25} />
      </Paint>
    </Oval>
  ))}
</CameraFrame>
```

```ts
const SLOTS = [0, 1, 2, 3]
const ovals = SLOTS.map(() => createHandle())

const detect = (frame: Frame) => {
  'worklet'
  const faces = detector.detectFaces(frame)
  for (const i of SLOTS) {
    const b = faces[i]?.bounds
    ovals[i]!.set({
      x: b?.x ?? 0,
      y: b?.y ?? 0,
      width: b?.width ?? 0,
      height: b?.height ?? 0,
    })
  }
}

const sink = createCameraFrameSink({
  onEachFrame: detect,
  onImage: (image) =>
    scene.add(
      cameraFrame({ image, ...sink.posture }, SLOTS.map((i) =>
        oval({ ref: ovals[i], x: 0, y: 0, width: 0, height: 0 }, [
          paint({}, [imageShader({ image }), blur({ blur: 25 })]),
        ])
      ))
    ),
})
```
